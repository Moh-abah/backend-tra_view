# المواصفة المعتمدة (v13.2) — المطبقة حرفياً:
#   R (المرجع)   = الشمعة صاحبة أكبر حجم في [i−liq_lookback, i−1] بشرط أنها
#                  حمراء (close<open). أكبر شمعة خضراء → لا مرجع (state 0).
#   ref_high     = أعلى high لشمعة حمراء داخل النافذة نفسه — محور الاختراق
#                  الموحد للحالات الثلاث (ليس بالضرورة قمة R).
#   G            = خضراء (close>open) + شرطان مشتركان:
#                  1) الانطلاقة: low_G ≥ price(0.382) لآخر ساق مؤكدة (أي اتجاه)
#                     — خط 0.618 بمقياس المستخدم (0 عند نهاية المسار).
#                  2) هيمنة الخضراء: volume_G > حجم كل الشموع الخضراء الواقعة
#                     حصرياً بين R وG (لا R ولا G ضمن المقارنة).
#   المصفوفة     : S1 = اختراق ∧ حجم_G > حجم_R
#                  S2 = اختراق ∧ حجم_G ≤ حجم_R   («ويبرغم عندها لفل حجم سيولتها»)
#                  S3 = لا اختراق ∧ حجم_G > حجم_R
#                  الرابعة (لا اختراق ∧ حجم أقل) = لا إشارة عمداً.
#   الدخول       = عند شمعة الإشارة نفسها (الحدث = الدخول؛ تقييم عند الإغلاق
#                  وتنفيذ عند افتتاح التالية — next_open) على أي اتجاه للساق.
#                  خطة بديلة وحيدة: إن لم تكن الساق مؤكدة لحظة الحدث → أول شمعة
#                  تحقق الانطلاقة خلال liq_arm_bars (FIFO — إسناد واحد للدخول).
#   SFP          = إشارة رابعة اختيارية مطفأة افتراضياً (liq_sfp_signals=false):
#                  اختراق بالضلع وإغلاق رجع تحت ref_high بحجم أكبر — تستبدل S1
#                  على شمعتها عند التفعيل (لا تضاعف إشارة أبداً). مطفأة = لا
#                  ذكر لها إطلاقاً وعدد الأعمدة يبقى 25.
